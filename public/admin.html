Firebase Setup

Connect to cloud database — data saved forever

⚡ How to get Firebase config (Free — 5 minutes)

  1. Go to console.firebase.google.com → Sign in with Google account
  2. Click "Create a project" → Name it eamcet-ace → Continue
  3. In left menu → click Firestore Database → Create database → choose Test mode (you'll replace the rules with the ones below anyway)
  4. In left menu → click Authentication → Get started → enable Email/Password → add yourself as a user (this is your admin login)
  5. Click the ⚙️ gear icon → Project Settings → scroll to "Your apps" → click </> Web icon → Register app → copy each firebaseConfig value below
  6. Click Save & Connect Firebase below — this is saved on this browser and read automatically by index.html too (same site, same storage), so there's nothing to copy into the other file.
  7. In left menu → Firestore Database → Rules tab → replace everything with the block below → Publish.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() { return request.auth != null; }
    function isStudent()  { return exists(/databases/$(database)/documents/students/$(request.auth.uid)); }
    function isAdmin()    { return isSignedIn() && request.auth.token.admin == true; }

    match /students/{studentId} {
      allow read: if isSignedIn();
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        isSignedIn() && request.auth.uid == studentId &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(
          ['totalScore','testsTaken','accuracySum','streak','lastActiveDate','subjectTotals','weakTopics','testBestScores','pendingVerification','setupLinkSent']
        ) &&
        request.resource.data.totalScore is number && request.resource.data.totalScore >= 0 &&
        request.resource.data.testsTaken is int && request.resource.data.testsTaken >= 0 &&
        request.resource.data.accuracySum is number && request.resource.data.accuracySum >= 0
      );
    }

    match /usernameLookup/{username} {
      allow get: if true;
      allow list: if false;
      allow write: if isAdmin();
    }

    match /questions/{id}       { allow read: if isSignedIn(); allow write: if isAdmin(); }
    match /resources/{id}       { allow read: if isSignedIn(); allow write: if isAdmin(); }
    match /mockTests/{id}       { allow read: if isSignedIn(); allow write: if isAdmin(); }
    match /settings/{id}        { allow read: if isSignedIn(); allow write: if isAdmin(); }
    match /dailyChallenges/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); }
    match /announcements/{id}   { allow read: if isSignedIn(); allow write: if isAdmin(); }

    match /results/{id} {
      allow create: if isSignedIn() &&
        request.resource.data.studentId == request.auth.uid &&
        request.resource.data.score is number &&
        request.resource.data.score >= 0 && request.resource.data.score <= 300;
      allow read: if isSignedIn();
      allow update, delete: if isAdmin();
    }
  }
}
